cmake_minimum_required(VERSION 3.23)
project(DreamUI VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dreamui_package STATIC src/zip_reader.cpp src/theme.cpp src/pe_probe.cpp src/control_hub.cpp src/address_db.cpp src/ink_cache.cpp src/native_ink.cpp src/ink_graph.cpp src/ink_load_patch.cpp src/dev/command_registry.cpp src/dev/command_inbox.cpp)
target_include_directories(dreamui_package PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
option(DREAMUI_EXPERIMENTAL_NATIVE_STAGE
  "Compile guarded native serializer-stage hook (requires explicit runtime opt-in)" OFF)
if(WIN32)
  add_library(DreamUI SHARED src/plugin.cpp src/win_probe.cpp src/win_load_hook.cpp)
  target_link_libraries(DreamUI PRIVATE dreamui_package version)
  if(DREAMUI_EXPERIMENTAL_NATIVE_STAGE)
    include(FetchContent)
    FetchContent_Declare(dreamui_minhook
      GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
      GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
    FetchContent_MakeAvailable(dreamui_minhook)
    target_link_libraries(DreamUI PRIVATE minhook)
    target_compile_definitions(DreamUI PRIVATE DREAMUI_EXPERIMENTAL_NATIVE_STAGE=1)
  endif()
  set_target_properties(DreamUI PROPERTIES PREFIX "" SUFFIX ".asi")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(package_tests tests/package_tests.cpp)
  target_link_libraries(package_tests PRIVATE dreamui_package)
  add_test(NAME package_reader COMMAND package_tests "${CMAKE_CURRENT_SOURCE_DIR}/themes/Default")
  add_executable(theme_tests tests/theme_tests.cpp)
  target_link_libraries(theme_tests PRIVATE dreamui_package)
  add_test(NAME native_theme_and_rollback COMMAND theme_tests "${CMAKE_CURRENT_SOURCE_DIR}/dist/Default.dream")
  add_executable(pe_probe_tests tests/pe_probe_tests.cpp)
  target_link_libraries(pe_probe_tests PRIVATE dreamui_package)
  add_test(NAME readonly_executable_probe COMMAND pe_probe_tests)
  add_executable(control_hub_tests tests/control_hub_tests.cpp)
  target_link_libraries(control_hub_tests PRIVATE dreamui_package)
  add_test(NAME dream_control_hub_model COMMAND control_hub_tests)

  add_executable(address_db_tests tests/address_db_tests.cpp)
  target_link_libraries(address_db_tests PRIVATE dreamui_package)
  add_test(NAME independent_address_db COMMAND address_db_tests "${CMAKE_CURRENT_BINARY_DIR}")
  add_executable(ink_cache_tests tests/ink_cache_tests.cpp)
  target_link_libraries(ink_cache_tests PRIVATE dreamui_package)
  add_test(NAME native_readonly_ink_cache COMMAND ink_cache_tests)
  add_executable(native_ink_lifecycle_tests tests/native_ink_lifecycle_tests.cpp)
  target_link_libraries(native_ink_lifecycle_tests PRIVATE dreamui_package)
  add_test(NAME typed_native_ink_lifecycle COMMAND native_ink_lifecycle_tests)
  add_executable(ink_graph_tests tests/ink_graph_tests.cpp)
  target_link_libraries(ink_graph_tests PRIVATE dreamui_package)
  add_test(NAME linked_ink_resource_graph COMMAND ink_graph_tests)
  add_executable(dev_commands_tests tests/dev_commands_tests.cpp)
  target_link_libraries(dev_commands_tests PRIVATE dreamui_package)
  add_test(NAME on_demand_dev_commands COMMAND dev_commands_tests)
endif()
